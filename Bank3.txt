pragma solidity ^0.8.20;

/*
 * Problem Statement: Write a smart contract on a test network, 
 * for Bank account of a customer for the following operations:
 * - Deposit money
 * - Withdraw money
 * - Show balance
 */

contract BankAccount {
    address public owner;
    uint256 public balance;

    // Events for transaction logging
    event Deposited(address indexed from, uint256 amount);
    event Withdrawn(address indexed to, uint256 amount);

    // Set the contract deployer as the account owner
    constructor() {
        owner = msg.sender;
    }

    // Function to deposit money into the bank account
    function deposit() external payable {
        require(msg.value > 0, "Deposit amount must be greater than zero");
        balance += msg.value;
        emit Deposited(msg.sender, msg.value);
    }

    // Function to withdraw money (restricted to the account owner)
    function withdraw(uint256 amount) external {
        require(msg.sender == owner, "Only the owner can withdraw funds");
        require(amount > 0 && amount <= balance, "Invalid amount or insufficient balance");
        
        // Update balance before transfer to prevent re-entrancy issues
        balance -= amount;
        
        // Transfer funds securely using low-level call
        (bool success, ) = payable(owner).call{value: amount}("");
        require(success, "Transfer failed");
        
        emit Withdrawn(owner, amount);
    }

    // Function to show the current balance of the account
    function showBalance() external view returns (uint256) {
        return balance;
    }
}

/*
 * Problem Statement: Write a program to solve a 0-1 Knapsack problem using 
 * dynamic programming or branch and bound strategy.
 */

#include <iostream>
#include <vector>
#include <algorithm>
#include <iomanip>

using namespace std;

// --- Method 1: Dynamic Programming Approach ---
void solveDynamicProgramming() {
    int n, W;
    cout << "\n--- 0-1 Knapsack (Dynamic Programming) ---\n";
    cout << "Number of items: "; cin >> n;
    cout << "Capacity: "; cin >> W;

    vector<int> wt(n), val(n);
    for (int i = 0; i < n; i++) {
        cout << "Weight and value of item " << i + 1 << ": ";
        cin >> wt[i] >> val[i];
    }

    // DP table where dp[i][w] stores maximum value with 'i' items and capacity 'w'
    vector<vector<int>> dp(n + 1, vector<int>(W + 1, 0));
    for (int i = 1; i <= n; i++) {
        for (int w = 0; w <= W; w++) {
            dp[i][w] = dp[i - 1][w];
            if (wt[i - 1] <= w) {
                dp[i][w] = max(dp[i][w], val[i - 1] + dp[i - 1][w - wt[i - 1]]);
            }
        }
    }

    cout << "Maximum value = " << dp[n][W] << '\n';
    cout << "Selected item numbers: ";
    int w = W;
    for (int i = n; i > 0; i--) {
        if (dp[i][w] != dp[i - 1][w]) {
            cout << i << " ";
            w -= wt[i - 1];
        }
    }
    cout << '\n';
}

// --- Method 2: Branch and Bound Approach ---
struct Item {
    double value, weight, ratio;
    int id;
};

void branchAndBoundRec(const vector<Item>& a, int i, double weight, int value, double cap, vector<int>& chosen, int& bestValue, vector<int>& bestSet) {
    if (weight > cap) return;
    if (i == (int)a.size()) {
        if (value > bestValue) {
            bestValue = value;
            bestSet = chosen;
        }
        return;
    }

    // Upper bound calculation using fractional knapsack greedy approach
    double bound = value, rem = cap - weight;
    for (int j = i; j < (int)a.size() && rem > 0; j++) {
        double take = min(rem, a[j].weight);
        bound += take * a[j].ratio;
        rem -= take;
    }

    // Prune branch if bound cannot exceed current best value
    if (bound <= bestValue) return;

    // Include item
    chosen.push_back(a[i].id);
    branchAndBoundRec(a, i + 1, weight + a[i].weight, value + (int)a[i].value, cap, chosen, bestValue, bestSet);
    chosen.pop_back();

    // Exclude item
    branchAndBoundRec(a, i + 1, weight, value, cap, chosen, bestValue, bestSet);
}

void solveBranchAndBound() {
    int n;
    double cap;
    cout << "\n--- 0-1 Knapsack (Branch and Bound) ---\n";
    cout << "Number of items: "; cin >> n;
    cout << "Capacity: "; cin >> cap;

    vector<Item> a(n);
    for (int i = 0; i < n; i++) {
        cout << "Value and weight of item " << i + 1 << ": ";
        cin >> a[i].value >> a[i].weight;
        if (a[i].weight <= 0) {
            cout << "Weight must be positive.\n";
            return;
        }
        a[i].ratio = a[i].value / a[i].weight;
        a[i].id = i + 1;
    }

    // Sort items by value-to-weight ratio descending for bounding
    sort(a.begin(), a.end(), [](const Item& x, const Item& y) {
        return x.ratio > y.ratio;
    });

    int bestValue = 0;
    vector<int> bestSet, chosen;
    branchAndBoundRec(a, 0, 0, 0, cap, chosen, bestValue, bestSet);

    cout << "Maximum value = " << bestValue << "\nSelected item numbers: ";
    for (int x : bestSet) cout << x << " ";
    cout << '\n';
}

int main() {
    int choice;
    cout << "Choose 0-1 Knapsack Method:\n";
    cout << "1. Dynamic Programming\n";
    cout << "2. Branch and Bound\n";
    cout << "Enter choice (1 or 2): ";
    cin >> choice;

    if (choice == 1) {
        solveDynamicProgramming();
    } else if (choice == 2) {
        solveBranchAndBound();
    } else {
        cout << "Invalid choice.\n";
    }

    return 0;
}

/*
 * Problem Statement: Implement K-Means clustering / hierarchical clustering on 
 * sales_data_sample.csv dataset. Determine the number of clusters using the elbow method.
 */

import os
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans, AgglomerativeClustering
from sklearn.metrics import silhouette_score
from sklearn.decomposition import PCA

# --- Step 1: Load and Pre-process Dataset ---
path = "sales_data_sample.csv"
if not os.path.exists(path):
    raise FileNotFoundError(f"Error: '{path}' not found. Please ensure sales_data_sample.csv is in the working directory.")

# Note: sales_data_sample.csv often uses latin1 or ISO-8859-1 encoding due to text formatting
df = pd.read_csv(path, encoding='latin1')

# Select only numeric columns and handle missing values
numeric = df.select_dtypes(include=np.number).dropna(axis=1, how="all").fillna(0)

if numeric.shape[1] < 2:
    raise ValueError("Error: Dataset must contain at least two numeric columns for clustering.")

# Scale the features
X = StandardScaler().fit_transform(numeric)

# --- Step 2: Determine Number of Clusters using Elbow Method & Silhouette Scores ---
inertias = []
silhouette_scores = []
K_range = range(2, 9)

print("--- Elbow Method & Silhouette Analysis ---")
for k in K_range:
    kmeans = KMeans(n_clusters=k, random_state=42, n_init=10)
    labels = kmeans.fit_predict(X)
    
    inertias.append(kmeans.inertia_)
    score = silhouette_score(X, labels)
    silhouette_scores.append(score)
    
    print(f"K = {k:2d} | Inertia (Elbow Metric) = {kmeans.inertia_:10.2f} | Silhouette Score = {score:.4f}")

# Select best k based on highest silhouette score (or inspect inertia drop for Elbow)
best_k = K_range[np.argmax(silhouette_scores)]
print(f"\nOptimal number of clusters selected (based on Silhouette Score): {best_k}\n")

# --- Step 3: Apply K-Means and Hierarchical Clustering ---
print("Applying K-Means and Agglomerative Hierarchical Clustering...")
km_labels = KMeans(n_clusters=best_k, random_state=42, n_init=10).fit_predict(X)
hc_labels = AgglomerativeClustering(n_clusters=best_k).fit_predict(X)

# --- Step 4: Visualize Results using PCA Decomposition ---
coords = PCA(n_components=2).fit_transform(X)

fig, axes = plt.subplots(1, 2, figsize=(12, 5))

# K-Means plot
axes[0].scatter(coords[:, 0], coords[:, 1], c=km_labels, cmap='viridis', edgecolor='k', s=50)
axes[0].set_title(f"K-Means Clustering (k={best_k})")
axes[0].set_xlabel("PCA Component 1")
axes[0].set_ylabel("PCA Component 2")

# Hierarchical clustering plot
axes[1].scatter(coords[:, 0], coords[:, 1], c=hc_labels, cmap='viridis', edgecolor='k', s=50)
axes[1].set_title(f"Agglomerative Hierarchical Clustering (k={best_k})")
axes[1].set_xlabel("PCA Component 1")
axes[1].set_ylabel("PCA Component 2")

plt.tight_layout()
plt.savefig("clustering_comparison.png", dpi=150)
print("Clustering comparison plot saved successfully as 'clustering_comparison.png'.")
plt.show()

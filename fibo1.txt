/*
 * Problem Statement: Write a non-recursive and recursive program 
 * to calculate Fibonacci numbers and analyze their time and space complexity.
 */

#include <iostream>
#include <chrono>

using namespace std;
using namespace chrono;

// Recursive approach: O(2^n) time, O(n) auxiliary stack space
long long fibRecursive(int n) {
    if (n <= 1) return n;
    return fibRecursive(n - 1) + fibRecursive(n - 2);
}

// Iterative (non-recursive) approach: O(n) time, O(1) space
long long fibIterative(int n) {
    if (n <= 1) return n;
    long long a = 0, b = 1;
    for (int i = 2; i <= n; ++i) {
        long long c = a + b;
        a = b;
        b = c;
    }
    return b;
}

int main() {
    int n;
    cout << "Enter n (0 to 45 for recursive timing): ";
    cin >> n;

    if (n < 0 || n > 45) {
        cout << "Error: Please enter n between 0 and 45.\n";
        return 1;
    }

    // Measure recursive execution time
    auto s1 = high_resolution_clock::now();
    long long r = fibRecursive(n);
    auto e1 = high_resolution_clock::now();

    // Measure iterative execution time
    auto s2 = high_resolution_clock::now();
    long long it = fibIterative(n);
    auto e2 = high_resolution_clock::now();

    // Output results and complexity analysis
    cout << "\n--- Fibonacci Calculation & Analysis ---\n";
    cout << "Recursive Result: " << r << "\n";
    cout << "Iterative Result: " << it << "\n\n";

    cout << "Execution Time:\n";
    cout << "  - Recursive: " << duration<double, milli>(e1 - s1).count() << " ms\n";
    cout << "  - Iterative: " << duration<double, milli>(e2 - s2).count() << " ms\n\n";

    cout << "Complexity Analysis:\n";
    cout << "  - Recursive: Time = O(2^n), Space = O(n) call stack\n";
    cout << "  - Iterative: Time = O(n),   Space = O(1)\n";

    return 0;
}
